workspace "CMS Microkernel Architecture" "Sơ đồ kiến trúc Microkernel cho CMS" {

    model {
        user = person "Quản trị viên" "Cài đặt và sử dụng các tính năng"

            seoPlugin = container "SEO Plug-in" "Module tối ưu hóa tìm kiếm."

            db = container "Database" "Lưu trữ dữ liệu hệ thống." "Database"
        }

        user -> core "Tương tác với hệ thống chính"
